cmake_minimum_required(VERSION 3.24)
project(Game LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(PainEngine CONFIG REQUIRED)
if(NOT TARGET PainEngine::Engine)
    message(FATAL_ERROR "This game requires a desktop SDK; a portable-core SDK is insufficient")
endif()
file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS "Source/*.cpp")
add_library(Game MODULE ${game_sources})
target_link_libraries(Game PRIVATE PainEngine::Engine)
set_target_properties(Game PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
pain_generate_components(Game Project "${CMAKE_CURRENT_SOURCE_DIR}/Source")
add_custom_command(TARGET Game POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_BINARY_DIR}/generated/Game/ProjectTypes.tsv" "$<TARGET_FILE_DIR:Game>/ProjectTypes.tsv")
